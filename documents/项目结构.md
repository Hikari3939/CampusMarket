# 项目结构

``` text
CampusMarket/
├── backend/                     # Flask 后端
│   ├── app/
│   │   ├── __init__.py          # Flask 应用工厂函数
│   │   ├── extensions.py        # 存放扩展 (db, jwt, cors, socketio)
│   │   ├── models.py            # SQLAlchemy ORM 模型 (User, Product, Order, Message)
│   │   ├── socket_events.py     # WebSocket 事件 (鉴权/收发消息/已读回执)
│   │   └── api/
│   │       ├── __init__.py
│   │       ├── auth.py          # 注册/登录（含密码强度校验）
│   │       ├── product.py       # 商品 CRUD（含分类筛选、分页、编辑）
│   │       ├── order.py         # 订单创建（行级锁）+ 取消（商品恢复）
│   │       ├── user.py          # 个人历史 + 资料修改 + 密码修改 + 公开主页
│   │       └── message.py       # 联系人列表 + 聊天历史
│   ├── static/                  # 本地静态资源映射
│   │   └── uploads/             # 用户上传的商品图片
│   ├── config.py                # 配置（从 .env 读取敏感信息）
│   ├── run.py                   # 项目启动入口（SocketIO + gevent）
│   ├── .env.example             # 环境变量模板
│   └── requirements.txt         # 依赖包列表
│
├── frontend/                    # Vue 3 前端
│   ├── public/
│   │   └── favicon.svg          # 网站图标
│   ├── .env.development         # 开发环境 API 地址
│   ├── .env.production          # 生产环境 API 地址
│   └── src/
│       ├── api/                 # 所有接口请求封装
│       │   ├── request.js       # Axios 拦截器 (Token 注入 + 401 处理 + 超时)
│       │   ├── auth.js          # 登录/注册
│       │   ├── product.js       # 商品 CRUD（含编辑 + 上传进度回调）
│       │   ├── order.js         # 订单创建 + 取消
│       │   ├── user.js          # 用户历史 + 资料修改 + 密码修改 + 公开主页
│       │   └── message.js       # 联系人 + 聊天历史
│       ├── stores/              # Pinia 全局状态
│       │   ├── user.js          # 用户登录态 + Token 管理
│       │   └── chat.js          # WebSocket 连接 + 消息/联系人管理
│       ├── router/
│       │   └── index.js         # 路由配置 + 导航守卫
│       ├── views/               # 页面级组件
│       │   ├── Home.vue         # 首页（分类筛选标签 + 骨架屏 + 无限滚动）
│       │   ├── Login.vue        # 登录/注册双 Tab（含密码规则提示）
│       │   ├── ProductDetail.vue # 商品详情（购买/编辑/联系 + 编辑对话框）
│       │   ├── Publish.vue      # 发布商品（分类选择 + 上传进度条）
│       │   ├── Profile.vue      # 个人中心（买卖历史 + 编辑资料 + 取消订单）
│       │   ├── Chat.vue         # 聊天中心（联系人列表 + 对话框）
│       │   └── UserProfile.vue  # 用户公开主页
│       ├── components/
│       │   └── ProductCard.vue  # 商品卡片（点击卖家名跳转主页）
│       ├── App.vue              # 根组件（WebSocket 生命周期 + 多标签页同步）
│       ├── main.js              # 入口（注册 Pinia / Router / ElementPlus）
│       └── style.css            # 全局主题（CSS 变量 + 东南大学 VIS 色彩）
│
├── documents/                   # 项目文档
│   ├── 项目文档.md              # 完整 API 文档 + 开发规范
│   ├── 项目结构.md              # 本文件
│   └── 开发流程.md              # 分阶段开发指南
│
├── mysql_init.sql               # 数据库初始化（建库 + 建表 + 索引）
└── Readme.md                    # 项目说明
```
