# 项目结构

backend/
├── app/
│   ├── __init__.py      # Flask 应用工厂函数
│   ├── extensions.py    # 存放扩展 (db, jwt, cors 等)，解决循环引用
│   ├── models.py        # SQLAlchemy ORM 模型 (映射数据库表)
│   ├── socket_events.py # WebSocket 核心事件监听与实时路由分发(连线、发消息、已读回执)
│   └── api/             # 存放所有接口蓝图 (Blueprints)
│       ├── __init__.py
│       ├── auth.py      # 登录、注册接口
│       ├── message.py   # 私信联系人列表与历史记录 RESTful 接口蓝图
│       ├── product.py   # 商品发布、查询、删除接口
│       ├── order.py     # 订单创建与交易防超卖接口
│       └── user.py      # 用户历史记录(我发布的/我买到的)查询接口
├── static/              # 本地静态资源映射目录
│   └── uploads/         # 存放用户上传的商品图片
├── config.py            # 配置文件 (数据库账号密码等)
├── run.py               # 项目启动入口
└── requirements.txt     # 依赖包列表

frontend/              # 项目根目录
├── .vscode/                 # VSCode 编辑器配置目录
├── node_modules/            # 自动生成的依赖包目录
├── public/                  # 静态资源目录
├── src/
│   ├── api/             # 存放所有接口请求
│   │   ├── request.js   # Axios 拦截器封装
│   │   ├── auth.js      # 登录注册接口调用
│   │   ├── message.js   # 私信相关 API 请求封装
│   │   ├── product.js   # 商品相关 API 请求封装
│   │   ├── order.js     # 订单相关 API 请求封装
│   │   └── user.js      # 用户历史记录 API 请求封装
│   ├── assets/        # 图片、字体等静态资源
│   ├── components/      # 复用组件
│   │   └── ProductCard.vue  # 商品复用卡片组件
│   ├── router/          # Vue Router 路由配置
│   │   └── index.js
│   ├── stores/          # Pinia 全局状态管理
│   │   ├── chat.js      # Pinia 全局实时通讯状态管理
│   │   └── user.js
│   ├── views/           # 页面级组件
│   │   ├── Chat.vue            # 实时聊天中心页面 (包含联系人列表与对话框双视图)
│   │   ├── Home.vue            # 首页
│   │   ├── Login.vue           # 登录/注册页
│   │   ├── Profile.vue         # 个人中心页 (展示买卖流水双 Tab 分栏)
│   │   ├── Publish.vue         # 发布商品页
│   │   └── ProductDetail.vue   # 商品详情页
│   ├── App.vue        # 根组件，作为入口挂载
│   ├── main.js          # JS 入口文件
│   └── style.css        # 全局主题与色彩变量
├── .gitignore               # Git 忽略文件设置
├── index.html               # 项目入口 HTML 文件
├── package-lock.json        # npm 自动生成的依赖锁文件
├── package.json             # 项目配置与依赖清单
├── README.md                # 项目说明文档
└── vite.config.js           # Vite 配置文件
